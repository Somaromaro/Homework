

فكرة البرنامج

- يتم إدخال الحروف فيه واحدًا تلو الآخر
- يتم إخراجها بنفس الترتيب
- عند الوصول لنهاية المصفوفة يرجع للبداية

---

شرح الكود

إنشاء المخزن

typedef struct {
    char b[N];
    int h;
    int t;
    int c;
} X;

- "b" → لتخزين الحروف
- "h" → مؤشر القراءة (head)
- "t" → مؤشر الكتابة (tail)
- "c" → عدد العناصر داخل المخزن

---

التهيئة

f0(&x);

تجعل المخزن فارغ وجميع المؤشرات تبدأ من 0

---

إدخال الاسم

scanf("%s", s);

---

إضافة النص المطلوب

strcat(s, "CE-ESY");

---

إدخال الحروف إلى المخزن

for (...) {
    f3(&x, s[i]);
}

- يتم إدخال كل حرف إلى المخزن
- في حال الامتلاء يتم طباعة: "Overflow"

---

دالة القراءة (f4)

char f4(X *x) {
    if (f2(x)) {
        printf("empty\n");
        return '\0';
    }

- تتحقق إذا كان المخزن فارغ
- إذا كان فارغ → تطبع "empty"
- إذا كان يحتوي بيانات:
  - تأخذ الحرف من الموقع "h"
  - تحرك "h" خطوة للأمام
  - تنقص عدد العناصر
  - تعيد الحرف

---

إخراج الحروف

while (!f2(&x)) {
    printf("%c", f4(&x));
}

- يتم إخراج الحروف واحدًا تلو الآخر
- يتم طباعة الناتج بنفس الترتيب

---

التأكد من أن المخزن فارغ

if (f2(&x)) {
    printf("empty now\n");
}

---

خاصية الدوران

x->t = (x->t + 1) % N;

- تعني أن المخزن دائري
- عند الوصول للنهاية يعود للبداية
